====================
File and procedure declaration contexts
====================

when ODIN_OS == .Windows {
    windows_only :: proc() {}
    foreign system {
        foreign_proc :: proc() ---
    }
} else when ODIN_OS == .Linux {
    linux_only :: proc() {}
} else {
    fallback :: proc() {}
}

container :: proc() {
    when FEATURE_ENABLED {
        local_proc :: proc() {}
        foreign local_library {
            local_foreign_proc :: proc() ---
        }
    }
}
