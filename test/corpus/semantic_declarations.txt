@(private)
compute :: #force_inline proc "contextless" (
    left: int,
    right := 1,
) -> (sum: int, ok: bool) where left > 0 #no_bounds_check {
    return left + right, true
}

Callback :: proc "c" (value: int) -> bool

foreign_api :: proc(input: int) -> int ---

accept_callback :: proc(callback: proc "c" (value: int) -> bool) {}

make_callback :: proc() -> proc(value: int) -> bool {
    return proc(value: int) -> bool {
        return value > 0
    }
}
